Int32 MyDialog::Message(const BaseContainer& msg, BaseContainer& result)
{
switch (msg.GetId())
{
case BFM_SCROLLGROUP_SCROLLED:
if (m_bHasHeader)
{
Int32 chgh = msg.GetInt32(0);
Int32 chgv = msg.GetInt32(1);
Int32 xoff = msg.GetInt32(2);
Int32 yoff = msg.GetInt32(3);
if (chgh)
m_wndHeaderArea.Redraw();
}
break;
}
return SUPER::Message(msg, result);
}

BFM_GETFOCUSBEFOREUPDATE 

A custom GUI receives this message before the layout is updated. This is used to avoid losing the focus over a specific gadget after the layout is updated.

See also
BFM_SETFOCUSAFTERUPDATE
Note
These messages are only sent if the customgui is used inside the Attribute Manager and not when it is used in any other GeDialog.
Int32 MyCustomGui::Message(const BaseContainer &msg, BaseContainer &result)
{
...
switch (msg.GetId())
{
case BFM_GETFOCUSBEFOREUPDATE: // Layout is about to update and '_activeGadgetId' could lose focus: store it
result.SetInt32(0, _activeGadgetId); // _activeGadgetId is a usual gadget id, e.g. AddButton(_activeGadgetId, BFH_LEFT, 0, 0 "My Button");
return true;
break;
case BFM_SETFOCUSAFTERUPDATE: // Layout has been updated (focus is in first gadget now): reactivate '_activeGadgetId'
_activeGadgetId = msg.GetInt32(0);
Activate(_activeGadgetId);
return true;
break;
...
}
return SUPER::Message(msg, result);
}
